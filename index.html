Steal this

The divider and the clip both move in pixel space via transform / clip-path — no layout, ever. The container's aspect-ratio is fixed, so dragging can't shift the page. touch-action: pan-y means horizontal drags belong to the slider while vertical scrolls still scroll the page.

let pos = .5, dragging = false, moved = 0, downT = 0, downX = 0, raf = 0;
function paint(){
  raf = 0;
  const x = pos * ba.clientWidth;
  divider.style.transform = `translateX(${x}px)`;
  before.style.clipPath = `inset(0 ${ba.clientWidth - x}px 0 0)`;
  ba.setAttribute('aria-valuenow', Math.round(pos * 100));
}
ba.addEventListener('pointerdown', e => {
  dragging = true; moved = 0; downX = e.clientX; downT = performance.now();
  ba.setPointerCapture(e.pointerId);
  ba.classList.add('dragging');
  hideHint();
});
ba.addEventListener('pointermove', e => {
  if(!dragging) return;
  moved = Math.max(moved, Math.abs(e.clientX - downX));
  pos = clamp((e.clientX - ba.getBoundingClientRect().left) / ba.clientWidth, .02, .98);
  if(!raf) raf = requestAnimationFrame(paint);
});
ba.addEventListener('pointerup', e => {
  // a tap (not a drag) glides the divider to the tap point
  if(moved < 6 && performance.now() - downT < 300) glideTo(e.clientX);
  dragging = false; ba.classList.remove('dragging');
});
